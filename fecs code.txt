// --- Server Handlers ---

void handleRoot() {
    server.send(200, "text/html", index_html);
}


void handleLogin() {

    // Check if all form data was sent
    if (
        server.hasArg("epa") &&
        server.hasArg("id") &&
        server.hasArg("code")
    ) {

        String inputEPA = server.arg("epa");
        String inputID = server.arg("id");
        String inputCode = server.arg("code");

        // Verify credentials
        if (
            inputEPA == validEPA &&
            inputID == validID &&
            inputCode == validCode
        ) {

            Serial.println("Login Successful! Turning on motor and GPS...");

            // Trigger the relay / motor driver
            digitalWrite(RELAY_PIN, HIGH);

            // Send success page
            server.send(200, "text/html", success_html);

        } else {

            Serial.println("Login Failed.");

            // Redirect back to login page with an error flag
            server.sendHeader("Location", "/?error=1");
            server.send(303);
        }

    } else {

        server.send(400, "text/plain", "Bad Request");
    }
}


// --- Setup ---

void setup() {

    Serial.begin(115200);

    // Configure relay pin
    pinMode(RELAY_PIN, OUTPUT);

    // Ensure motor is OFF by default
    digitalWrite(RELAY_PIN, LOW);

    // Set up ESP32 as an Access Point
    Serial.println("Setting up Access Point...");

    WiFi.softAP(ssid, password);

    // Get Access Point IP address
    IPAddress IP = WiFi.softAPIP();

    Serial.print("AP IP address: ");
    Serial.println(IP);

    // Define server routes
    server.on("/", HTTP_GET, handleRoot);
    server.on("/login", HTTP_POST, handleLogin);

    // Start web server
    server.begin();

    Serial.println("Web server started.");
}


// --- Main Loop ---

void loop() {

    // Listen for incoming clients
    server.handleClient();
}
