// Read all available NMEA data from the GPS module.
void readGPS() {
    while (gpsSerial.available() > 0) {
        gps.encode(gpsSerial.read());
    }
}

void handleRoot() {
    server.send(200, "text/html", index_html);
}

void handleGPS() {
    if (!gpsActivated) {
        server.send(403, "application/json", "{\"error\":\"GPS is not activated\"}");
        return;
    }

    String json = "{\"active\":true,\"valid\":";
    json += gps.location.isValid() ? "true" : "false";
    json += ",\"latitude\":";
    json += gps.location.isValid() ? String(gps.location.lat(), 6) : "0";
    json += ",\"longitude\":";
    json += gps.location.isValid() ? String(gps.location.lng(), 6) : "0";
    json += ",\"satellites\":";
    json += gps.satellites.isValid() ? String(gps.satellites.value()) : "0";
    json += ",\"age_ms\":";
    json += gps.location.isValid() ? String(gps.location.age()) : "-1";
    json += "}";

    server.send(200, "application/json", json);
}

void handleLogin() {
    if (!server.hasArg("epa") || !server.hasArg("id") || !server.hasArg("code")) {
        server.send(400, "text/plain", "Bad Request");
        return;
    }

    if (server.arg("epa") == validEPA &&
        server.arg("id") == validID &&
        server.arg("code") == validCode) {

        Serial.println("Login successful. Activating GPS and unlocking motor...");
        gpsActivated = true;
        digitalWrite(RELAY_PIN, HIGH);
        server.send(200, "text/html", success_html);
    } else {
        Serial.println("Login failed.");
        server.sendHeader("Location", "/?error=1");
        server.send(303);
    }
}

void setup() {
    Serial.begin(115200);

    pinMode(RELAY_PIN, OUTPUT);
    digitalWrite(RELAY_PIN, LOW);

    // Start UART communication with the GPS module.
    gpsSerial.begin(GPS_BAUD_RATE, SERIAL_8N1, GPS_RX_PIN, GPS_TX_PIN);

    Serial.println("Setting up Access Point...");
    WiFi.softAP(ssid, password);
    Serial.print("AP IP address: ");
    Serial.println(WiFi.softAPIP());

    server.on("/", HTTP_GET, handleRoot);
    server.on("/login", HTTP_POST, handleLogin);
    server.on("/gps", HTTP_GET, handleGPS);
    server.begin();

    Serial.println("Web server started.");
    Serial.println("GPS wiring: GPS TX -> GPIO16, GPS RX -> GPIO17");
}

void loop() {
    // Keep parsing GPS data even while no user is logged in, so a fix is ready quickly.
    readGPS();
    server.handleClient();
}
